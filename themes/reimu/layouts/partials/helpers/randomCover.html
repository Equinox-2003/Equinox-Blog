{{- $covers := slice -}} {{- with .Site.Data.covers -}} {{- if reflect.IsSlice . -}} {{- range . -}} {{- $cover := strings.TrimSpace (printf "%v" .) -}} {{- if $cover -}} {{- $covers = $covers | append $cover -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $cover := "" -}} {{- if gt (len $covers) 0 -}} {{- /* Maintain a shuffled queue so covers only repeat after a full cycle. */ -}} {{- $scratchKey := "helpers/randomCover/queue" -}} {{- $queue := .Scratch.Get $scratchKey | default (slice) -}} {{- if lt (len $queue) 1 -}} {{- $queue = shuffle $covers -}} {{- end -}} {{- $cover = index $queue 0 -}} {{- $nextQueue := cond (gt (len $queue) 1) (after 1 $queue) (slice) -}} {{- .Scratch.Set $scratchKey $nextQueue -}} {{- end -}} {{- return $cover -}}
